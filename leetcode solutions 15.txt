reverse integer

class Solution {

    public int reverse(int x) {

        // Здесь будем хранить перевёрнутое число.
        int result = 0;

        // Пока в x ещё остались цифры,
        // продолжаем доставать последнюю цифру.
        while (x != 0) {

            // Получаем последнюю цифру числа.
            //
            // Например:
            // x = 123
            // x % 10 = 3
            //
            // Для отрицательного числа:
            // x = -123
            // x % 10 = -3
            int digit = x % 10;


            // Убираем последнюю цифру из x.
            //
            // Например:
            // 123 / 10 = 12
            // 12 / 10 = 1
            // 1 / 10 = 0
            //
            // В Java деление int отбрасывает дробную часть.
            x = x / 10;


            // ПЕРЕД тем как сделать:
            //
            // result = result * 10 + digit
            //
            // проверяем, не произойдёт ли overflow.
            //
            // Это необходимо, потому что int в Java
            // может хранить только:
            //
            // -2147483648 ... 2147483647

            if (result > Integer.MAX_VALUE / 10 ||
                (result == Integer.MAX_VALUE / 10 && digit > 7)) {

                // Если число станет больше 2147483647,
                // возвращаем 0 согласно условию задачи.
                return 0;
            }


            if (result < Integer.MIN_VALUE / 10 ||
                (result == Integer.MIN_VALUE / 10 && digit < -8)) {

                // Если число станет меньше -2147483648,
                // возвращаем 0.
                return 0;
            }


            // Добавляем новую цифру в конец result.
            //
            // Например:
            //
            // result = 12
            // digit = 3
            //
            // result = 12 * 10 + 3
            //        = 123
            result = result * 10 + digit;
        }


        // Возвращаем перевёрнутое число.
        return result;
    }
}
